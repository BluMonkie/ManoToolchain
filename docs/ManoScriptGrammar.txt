program = { decl | stmt } ;

decl = vardecl ;
vardecl = "var" , IDENTIFIER , ":" , IDENTIFIER , [ "=" , expr] , ";" ;

stmt = asm | block | if_stmt | expr_stmt ;
asm = "asm" , "{" , { ASM_INST } , "}" ;
block = "{" , { stmt | vardecl } , "}" ;
if_stmt = "if" , expr , block , [ "else" , ( block | if_stmt ) ] ;
expr_stmt = expr , ";" ;

expr = assignment ;
assignment = ( IDENTIFIER , "=" , assignment ) | comparison ;
comparison = bitwise , [ ( "<" | "<=" | ">" | ">=" | "==" | "!=" ) , bitwise ] ;
bitwise = shift , { ( "|" | "&" | "^" ) , shift } ;
shift = term , { ( "<<" | ">>" ) , term } ;
term = unary , { ( "+" | "-" ) , unary } ;
unary = ( "+" | "-" | "~" ) , unary | primary;
primary = "(" , expr , ")" | LITERAL | IDENTIFIER ;
